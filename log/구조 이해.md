🔄 3. useEffect - 서버 연결 테스트
javascriptuseEffect(() => {
  fetch('http://localhost:5000/')
    .then(res => res.json())
    .then(data => {
      setServerStatus('✅ 연결됨');
    })
    .catch(err => {
      setServerStatus('❌ 연결 실패');
    });
}, []);
단계별 동작:

앱이 처음 시작될 때 자동 실행 ([] = 한 번만 실행)
fetch(): 서버에 요청 보내기

javascript   fetch('http://localhost:5000/')

응답 처리:

javascript   .then(res => res.json())  // 응답을 JSON으로 변환
   .then(data => {
     // 성공! 서버 상태를 '연결됨'으로 변경
     setServerStatus('✅ 연결됨');
   })

오류 처리:

javascript   .catch(err => {
     // 실패! 서버 상태를 '연결 실패'로 변경
     setServerStatus('❌ 연결 실패');
   })
비유:

전화를 걸어서(fetch) 상대방이 받으면(.then) "연결됨", 안 받으면(.catch) "연결 실패"

4.6 runBacktest - 백테스트 실행
javascriptconst runBacktest = async () => {
  setLoading(true);  // 로딩 시작
  
  try {
    // 1. 서버에 요청 보내기
    const response = await fetch('http://localhost:5000/api/backtest/run', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        portfolio: {
          name: portfolioName,
          holdings,
          initialCapital
        },
        startDate,
        endDate
      })
    });

    // 2. 성공 응답 처리
    if (response.ok) {
      const result = await response.json();
      setBacktestResult({
        ...result,
        portfolioName,
        holdings,
        settings: { startDate, endDate, initialCapital }
      });
      setStep(5);  // 결과 화면으로
    } else {
      throw new Error('서버 응답 오류');
    }
  } catch (error) {
    // 3. 실패 시 모의 데이터 사용
    console.log('서버 연결 실패, 모의 데이터 사용:', error);
    
    await new Promise(resolve => setTimeout(resolve, 2000));  // 2초 대기
    
    const mockResult = {
      performance: {
        totalReturn: (Math.random() * 100 - 20).toFixed(2),
        // ... 랜덤 데이터
      },
      portfolioName,
      holdings,
      settings: { startDate, endDate, initialCapital }
    };
    
    setBacktestResult(mockResult);
    setStep(5);
    alert('⚠️ 서버에서 실제 데이터를 가져올 수 없어 모의 데이터를 표시합니다.');
  } finally {
    setLoading(false);  // 로딩 종료 (성공이든 실패든)
  }
};
